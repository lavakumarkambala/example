#define PDF_ERROR 919051

}
int registerMGExt(int*, va_list)
{
	printf("\n ProjectPDF extension registered...\n");
	printError(EPM_register_rule_handler("MG CHECK PDF",
		"this handler is to check PDF", (EPM_rule_handler_t)MGCheckPDF));
	return 0;
}
EPM_decision_t MGCheckPDF(EPM_rule_message_t msg)
{
	EPM_decision_t decision = EPM_nogo;
	int count = 0;
	tag_t root_task = msg.task;
	tag_t* attachments = NULL;
	int record = 0;
	printError(EPM_ask_attachments(root_task, EPM_target_attachment, &count, &attachments));
	char* userID = NULL;
	for (int i = 0; i < count; i++)
	{
		char* value = NULL;
		int countt = 0;
		tag_t* secondary_objects;
		(GRM_list_secondary_objects_only(attachments[i], NULLTAG, &countt, &secondary_objects));
		for (int j = 0; j < countt; j++)
		{
			printError(AOM_ask_value_string(secondary_objects[j], "object_type", &value));
			if (tc_strcmp(value, "PDF") == 0)
			{
				record++;
			}
		}
		if (secondary_objects) MEM_free(secondary_objects);
	}
	if (count == record)
	{
		decision = EPM_go;
	}
	if (decision == 0)
	{
		printError(POM_get_user_id(&userID));
		printError(EMH_store_error_s1(EMH_severity_error, PDF_ERROR, userID));
		printError(PDF_ERROR);
	}
	if (attachments) MEM_free(attachments);
	//if (userID) MEM_free(userID);
	return decision;
}
